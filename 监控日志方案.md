## 二、SDK 核心实现

### 初始化 + 会话管理

```js
class Tracker {
  constructor(config) {
    this.appId = config.appId;
    this.userId = null;
    this.sessionId = this.getOrCreateSession();
    this.anonymousId = this.getOrCreateAnonymousId();
    this.queue = [];

    this.initPageView();
    this.initClick();
    this.initRequest();
    this.initReporter();
  }

  getOrCreateAnonymousId() {
    let id = localStorage.getItem('_anon_id');
    if (!id) {
      id = 'anon_' + Date.now() + '_' + Math.random().toString(36).slice(2);
      localStorage.setItem('_anon_id', id);
    }
    return id;
  }

  getOrCreateSession() {
    // session 30分钟无操作过期
    const KEY = '_session';
    const EXPIRE = 30 * 60 * 1000;
    const stored = JSON.parse(sessionStorage.getItem(KEY) || 'null');

    if (stored && Date.now() - stored.lastActive < EXPIRE) {
      stored.lastActive = Date.now();
      sessionStorage.setItem(KEY, JSON.stringify(stored));
      return stored.id;
    }

    const session = {
      id: 'sess_' + Date.now() + '_' + Math.random().toString(36).slice(2),
      lastActive: Date.now()
    };
    sessionStorage.setItem(KEY, JSON.stringify(session));
    return session.id;
  }

  // 构建基础字段，每条日志都带
  baseFields() {
    return {
      appId: this.appId,
      userId: this.userId,
      anonymousId: this.anonymousId,
      sessionId: this.sessionId,
      url: location.href,
      path: location.pathname,
      referrer: document.referrer,
      timestamp: Date.now(),
      userAgent: navigator.userAgent,
    };
  }

  track(eventType, properties = {}) {
    const event = {
      eventId: crypto.randomUUID(),
      eventType,
      ...this.baseFields(),
      properties,
    };
    this.queue.push(event);
  }
}
```

### 1. 页面访问采集

```js
initPageView() {
  let startTime = Date.now();
  let lastPath = location.pathname;

  const trackView = () => {
    this.track('page_view', {
      title: document.title,
      path: location.pathname,
    });
    startTime = Date.now();
    lastPath = location.pathname;
  };

  // 记录离开时的停留时长，打到上一条 page_view 里
  const trackLeave = () => {
    this.track('page_leave', {
      path: lastPath,
      duration: Date.now() - startTime,  // 停留时长(ms)
    });
  };

  // 首次进入
  trackView();

  // SPA 路由跳转
  const originPush = history.pushState;
  const originReplace = history.replaceState;

  history.pushState = (...args) => {
    trackLeave();
    originPush.apply(history, args);
    trackView();
  };
  history.replaceState = (...args) => {
    trackLeave();
    originReplace.apply(history, args);
    trackView();
  };

  window.addEventListener('popstate', () => {
    trackLeave();
    trackView();
  });

  // 页面隐藏/关闭
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) trackLeave();
    else startTime = Date.now(); // 重新激活，重置计时
  });
}
```

### 2. 点击采集

```js
initClick() {
  document.addEventListener('click', (e) => {
    const el = e.target;

    this.track('click', {
      // 元素信息
      tagName: el.tagName,
      text: el.innerText?.slice(0, 100),
      id: el.id || null,
      className: el.className || null,
      // 业务埋点标识（推荐在元素上加 data-track-id）
      trackId: el.closest('[data-track-id]')?.dataset.trackId || null,
      // 点击坐标
      x: e.clientX,
      y: e.clientY,
      // 元素路径，方便还原点击位置
      xpath: getXPath(el),
    });
  }, true); // 捕获阶段，防止冒泡被阻止
}

// 生成 XPath：div > ul > li:nth-child(2) > button
function getXPath(el) {
  const parts = [];
  while (el && el.nodeType === Node.ELEMENT_NODE) {
    let index = 0;
    let sibling = el;
    while (sibling) { if (sibling.nodeName === el.nodeName) index++; sibling = sibling.previousElementSibling; }
    parts.unshift(`${el.nodeName.toLowerCase()}:nth-child(${index})`);
    el = el.parentNode;
    if (parts.length > 5) break; // 最多5层，避免太长
  }
  return parts.join(' > ');
}
```

### 3. 请求采集（拦截 fetch + XHR）

```js
initRequest() {
  this.interceptFetch();
  this.interceptXHR();
}

interceptFetch() {
  const originalFetch = window.fetch;
  window.fetch = async (input, init = {}) => {
    const url = typeof input === 'string' ? input : input.url;
    const method = (init.method || 'GET').toUpperCase();
    const startTime = Date.now();

    try {
      const response = await originalFetch(input, init);
      this.track('api_request', {
        url: this.sanitizeUrl(url),   // 脱敏处理
        method,
        status: response.status,
        duration: Date.now() - startTime,
        success: response.ok,
      });
      return response;
    } catch (err) {
      this.track('api_request', {
        url: this.sanitizeUrl(url),
        method,
        status: 0,
        duration: Date.now() - startTime,
        success: false,
        error: err.message,
      });
      throw err;
    }
  };
}

interceptXHR() {
  const OriginalXHR = window.XMLHttpRequest;
  window.XMLHttpRequest = function () {
    const xhr = new OriginalXHR();
    let method, url, startTime;

    const originalOpen = xhr.open;
    xhr.open = function (m, u, ...rest) {
      method = m.toUpperCase();
      url = u;
      return originalOpen.apply(xhr, [m, u, ...rest]);
    };

    const originalSend = xhr.send;
    xhr.send = function (...args) {
      startTime = Date.now();
      xhr.addEventListener('loadend', () => {
        tracker.track('api_request', {
          url: tracker.sanitizeUrl(url),
          method,
          status: xhr.status,
          duration: Date.now() - startTime,
          success: xhr.status >= 200 && xhr.status < 300,
        });
      });
      return originalSend.apply(xhr, args);
    };
    return xhr;
  };
}

// 去掉 URL 中的敏感参数
sanitizeUrl(url) {
  try {
    const u = new URL(url, location.origin);
    ['token', 'password', 'phone', 'idcard'].forEach(k => {
      if (u.searchParams.has(k)) u.searchParams.set(k, '***');
    });
    return u.pathname + u.search; // 只保留路径+参数，去掉域名
  } catch {
    return url;
  }
}
```

### 4. 上报模块

```js
initReporter() {
  // 定时上报
  setInterval(() => this.flush(), 5000);

  // 页面隐藏时上报（配合 page_leave）
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) this.flush();
  });
}

flush() {
  if (!this.queue.length) return;
  const events = this.queue.splice(0); // 取出并清空

  const payload = JSON.stringify({ events });

  // 优先 sendBeacon（页面关闭不丢数据）
  if (navigator.sendBeacon) {
    navigator.sendBeacon('/api/track', payload);
    return;
  }
  fetch('/api/track', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: payload,
    keepalive: true,
  });
}
```

## 三、服务端日志结构（存 ClickHouse）

```sql
CREATE TABLE user_events (
  event_id     String,
  event_type   String,        -- page_view / click / api_request / page_leave
  app_id       String,
  user_id      Nullable(String),
  anonymous_id String,
  session_id   String,

  -- 页面信息
  url          String,
  path         String,
  referrer     String,

  -- 事件属性（JSON）
  properties   String,        -- 存 JSON，灵活扩展

  timestamp    DateTime64(3), -- 毫秒精度
  date         Date           -- 分区键
)
ENGINE = MergeTree()
PARTITION BY date
ORDER BY (app_id, user_id, timestamp);
```

## 四、查询示例

### 查询某用户一段时间内的操作轨迹

```sql
SELECT
  timestamp,
  event_type,
  path,
  JSONExtractString(properties, 'title')    AS page_title,
  JSONExtractString(properties, 'text')     AS click_text,
  JSONExtractString(properties, 'trackId')  AS track_id,
  JSONExtractString(properties, 'url')      AS api_url,
  JSONExtractString(properties, 'method')   AS api_method,
  JSONExtractInt(properties, 'status')      AS api_status,
  JSONExtractInt(properties, 'duration')    AS duration_ms
FROM user_events
WHERE user_id = 'u_123'
  AND timestamp BETWEEN '2026-05-01' AND '2026-05-18'
ORDER BY timestamp ASC;
```

### 统计页面访问量（PV/UV）

```sql
SELECT
  path,
  count()                           AS pv,
  count(DISTINCT user_id)           AS uv,
  avg(JSONExtractInt(properties, 'duration')) AS avg_stay_ms
FROM user_events
WHERE event_type = 'page_view'
  AND date = '2026-05-18'
GROUP BY path
ORDER BY pv DESC;
```

### 统计按钮点击热度

```sql
SELECT
  JSONExtractString(properties, 'trackId') AS track_id,
  path,
  count() AS clicks
FROM user_events
WHERE event_type = 'click'
  AND date = '2026-05-18'
GROUP BY track_id, path
ORDER BY clicks DESC
LIMIT 20;
```

### 查慢请求 + 错误请求

```sql
SELECT
  JSONExtractString(properties, 'url')    AS api,
  JSONExtractString(properties, 'method') AS method,
  JSONExtractInt(properties, 'status')    AS status,
  count()                                 AS count,
  avg(JSONExtractInt(properties, 'duration')) AS avg_ms,
  max(JSONExtractInt(properties, 'duration')) AS max_ms
FROM user_events
WHERE event_type = 'api_request'
  AND date = '2026-05-18'
GROUP BY api, method, status
HAVING avg_ms > 1000 OR status >= 400  -- 慢请求或报错
ORDER BY count DESC;
```

## 五、整体数据流

```
浏览器 SDK
  ├── page_view  ─┐
  ├── click      ─┼──> 批量队列 ──> sendBeacon ──> /api/track
  └── api_request─┘         (5s 或 10条 或 页面隐藏)
                                          │
                                     Kafka Topic
                                          │
                               ┌──────────┴──────────┐
                           实时流处理               离线批处理
                           Flink                    Hive/Spark
                           (实时看板)               (历史分析)
                                │                       │
                           ClickHouse              数据仓库
                                │
                           查询 API ──> 运营后台 / 告警系统
```
